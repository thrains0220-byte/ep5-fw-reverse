def hexdump(data):
    from string import digits, letters, punctuation
    for i in range(0,len(data), 16):
        print "%04x" % (i,), 
        print "", " ".join(x.encode("hex") for x in data[i:i+16]).ljust(32+15), 
        print "", repr("".join(x in (digits+letters+punctuation+" ") and x or "." for x in data[i:i+16]))

import struct
class oly:
    def __init__(self, dev):
        self.dev = dev
        import os
        assert os.access(dev, os.W_OK)
        self.idx = 0
        self.code = None

    def c0(self, code, *args):
        self.idx += 1
        self.code = code
        cmd = struct.pack(">B 7x H 6x", 0xc0, 0x40)
        data = struct.pack("< %sL" % (len(args),), *args)
        data = struct.pack("<L BxH L", 0x0c+len(data), 0x1, code, self.idx) + data
        data += "\x00" * (0x40-len(data))
        if debug > 30:
            print
            print "-"*10
            print "-> 0xc0", hex(0x1), hex(code), hex(self.idx), map(hex, args)
        scsi(self.dev, cmd, data)

    def c2(self, size):
        cmd = struct.pack(">B 7x H 6x", 0xc2, size)
        if debug > 30:
            print "-> 0xc2", hex(size)
        data = scsi(self.dev, cmd, rsize=size)
        assert len(data) == size
        size2, type, code, idx = struct.unpack("<L BxH L", data[:12])
        assert idx == self.idx
        assert type == 0x2 
        assert code == self.code
        assert size2 == size

        ret = data[12:size]
        if debug > 30:
            print "<-     ", hex(type), hex(code), hex(idx)
            hexdump(ret)
        return ret

    def c3(self):
        cmd = struct.pack(">B 7x H 6x", 0xc3, 0x40)
        if debug > 30:
            print "-> 0xc3"
        data = scsi(self.dev, cmd, rsize=0x40)
        assert len(data) == 0x40
        size, type, code, idx = struct.unpack("<L BxH L", data[:12])
        assert idx == self.idx
        assert type == 0x3 
        assert code == 0x2001 or (self.code == 0x1002 and code == 0x201e) or (self.code == 0x9124 and code == 0x201d)
        assert size % 4 == 0
        ret = struct.unpack("<%sL" % (size/4-3,), data[12:size])
        if debug > 30:
            print "<-     ", hex(type), hex(code), hex(idx), map(hex, ret)
        return ret

    def c4(self):
        cmd = struct.pack(">B 7x H 6x", 0xc4, 0x40)
        if debug > 30:
            print "-> 0xc4"
        data = scsi(self.dev, cmd, rsize=0x40)
        assert len(data) == 0x40
        size, type, code, idx = struct.unpack("<L BxH L", data[:12])
        assert idx == self.idx
        assert type == 0x2 
        assert code == 0x2001

        assert size == 0x10
        ret = struct.unpack("<L", data[12:size])
        if debug > 30:
            print "<-     ", hex(type), hex(code), hex(idx), map(hex,ret)
        return ret
